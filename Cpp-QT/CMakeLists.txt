cmake_minimum_required(VERSION 3.22)
project(MyDictQt VERSION 1.7.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Widgets Network Sql Concurrent Svg)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Widgets Network Sql Concurrent Svg)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(mydict_core STATIC src/core.cpp src/core.h src/services.cpp src/services.h src/dailyword.cpp src/dailyword.h src/sync.cpp src/sync.h src/importer.cpp src/importer.h resources.qrc)
target_include_directories(mydict_core PUBLIC src)
target_link_libraries(mydict_core PUBLIC Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Gui Qt${QT_VERSION_MAJOR}::Network Qt${QT_VERSION_MAJOR}::Sql Qt${QT_VERSION_MAJOR}::Concurrent)
target_compile_definitions(mydict_core PUBLIC MYDICT_VERSION="${PROJECT_VERSION}-shared")
add_library(mydict_ui STATIC src/birdpet.cpp src/birdpet.h src/mainwindow.cpp src/mainwindow.h src/dailypage.cpp src/dailypage.h src/entryrenderer.cpp src/entryrenderer.h src/appfonts.cpp src/appfonts.h)
target_link_libraries(mydict_ui PUBLIC mydict_core Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Svg)
if(WIN32)
  target_link_libraries(mydict_ui PUBLIC dwmapi)
endif()
add_executable(MyDictQt WIN32 MACOSX_BUNDLE src/main.cpp)
if(WIN32)
  target_sources(MyDictQt PRIVATE app.rc)
elseif(APPLE)
  set(MYDICT_MACOS_ICON "${CMAKE_CURRENT_BINARY_DIR}/app-icon.icns")
  add_custom_command(
    OUTPUT "${MYDICT_MACOS_ICON}"
    COMMAND /bin/bash "${CMAKE_CURRENT_SOURCE_DIR}/tools/macos-icon.sh"
            "${CMAKE_CURRENT_SOURCE_DIR}/assets/app-icon.png" "${MYDICT_MACOS_ICON}"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app-icon.png"
            "${CMAKE_CURRENT_SOURCE_DIR}/tools/macos-icon.sh"
    COMMENT "Generating the macOS Finder and Dock icon"
    VERBATIM)
  # Generate before the bundle's resource-copy phase, including a clean first build.
  add_custom_target(MyDictMacIcon DEPENDS "${MYDICT_MACOS_ICON}")
  add_dependencies(MyDictQt MyDictMacIcon)
  set_source_files_properties("${MYDICT_MACOS_ICON}" PROPERTIES
    GENERATED TRUE MACOSX_PACKAGE_LOCATION "Resources")
  target_sources(MyDictQt PRIVATE "${MYDICT_MACOS_ICON}")
  set_target_properties(MyDictQt PROPERTIES MACOSX_BUNDLE_ICON_FILE "app-icon.icns")
endif()
target_link_libraries(MyDictQt PRIVATE mydict_ui)
set_target_properties(MyDictQt PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "拾词 MyDict" MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.hippo-yf.mydict" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
add_executable(MyDictTool src/tool.cpp)
target_link_libraries(MyDictTool PRIVATE mydict_core)
if(BUILD_TESTING)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
  foreach(test storage services ui sync importer fonts daily birdpet)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${test}_test.cpp")
      add_executable(${test}_test tests/${test}_test.cpp)
      target_link_libraries(${test}_test PRIVATE mydict_ui Qt${QT_VERSION_MAJOR}::Test)
      if(test STREQUAL "fonts" OR test STREQUAL "ui")
        target_compile_definitions(${test}_test PRIVATE FONT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fonts")
      endif()
      add_test(NAME ${test} COMMAND ${test}_test)
      set_tests_properties(${test} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
    endif()
  endforeach()
  if(WIN32)
    add_executable(native_fonts_test tests/native_fonts_test.cpp)
    target_link_libraries(native_fonts_test PRIVATE mydict_ui Qt${QT_VERSION_MAJOR}::Test)
    target_compile_definitions(native_fonts_test PRIVATE FONT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fonts")
    foreach(scale 100 150)
      if(scale EQUAL 100)
        set(font_scale "1")
      else()
        set(font_scale "1.5")
      endif()
      add_test(NAME native_fonts_${scale} COMMAND native_fonts_test
        -o "${CMAKE_BINARY_DIR}/native-fonts-${scale}.txt,txt")
      set_tests_properties(native_fonts_${scale} PROPERTIES TIMEOUT 120
        ENVIRONMENT "QT_QPA_PLATFORM=windows:fontengine=freetype;QT_SCALE_FACTOR=${font_scale};MYDICT_NATIVE_FONT_QA_OUTPUT_DIR=${CMAKE_BINARY_DIR}/native-fonts-${scale}")
    endforeach()
  endif()
endif()
install(TARGETS MyDictQt MyDictTool BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(APPLE)
  set(MYDICT_ASSETS "MyDictQt.app/Contents/Resources")
else()
  set(MYDICT_ASSETS "bin")
endif()
install(FILES settings.qt.example.json DESTINATION "${MYDICT_ASSETS}" RENAME settings.example.json)
install(FILES LICENSE QT-NOTICES.md DESTINATION "${MYDICT_ASSETS}")
install(DIRECTORY packaging/licenses/ DESTINATION "${MYDICT_ASSETS}/third-party-licenses")
install(DIRECTORY fonts/ DESTINATION "${MYDICT_ASSETS}/fonts" PATTERN "local" EXCLUDE)
install(FILES packaging/mydict.desktop DESTINATION share/applications)
install(FILES assets/app-icon.png DESTINATION share/icons/hicolor/256x256/apps RENAME mydict.png)
set(CPACK_PACKAGE_NAME MyDictQt)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_VENDOR "Yance Feng & Codex")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/packages")
if(WIN32)
  set(CPACK_GENERATOR ZIP)
elseif(APPLE)
  set(CPACK_GENERATOR DragNDrop)
else()
  set(CPACK_GENERATOR TGZ)
endif()
include(CPack)
